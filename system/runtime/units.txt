# The SUN runtime: what a native program links on top of the POSIX runtime
# (docs/userland/NATIVE_APPS.md, docs/userland/SUN.md).
#
# Read by CMake (SAVANXP_SUN_RUNTIME_SOURCES, linked into every NATIVE
# program) and by tools/make-sdk.sh, which prebuilds the units into the SDK
# sysroot as the groups sun-core and sun-gui. Same format as the POSIX
# manifest, subsystems/posix/sdk/v1/runtime/units.txt: "<group> <path from the
# repository root>", in link order.
#
#   core    the heap's page source on SXN (docs/userland/NATIVE_HEAP.md),
#           replacing the POSIX runtime's weak one, the person's profile
#           and settings path (include/sxn/sxnprofile.h), the DNS resolver
#           (include/sxn/sxndns.h), the HTTP loader (include/sxn/sxnhttp.h) and
#           the .sxe reader's file (system/runtime/sxn_sxe.c)
#   gui     the SXGUI application frame for SUN (include/sxn/sxngui.h), whose
#           strong toolkit host replaces the POSIX frame's weak one, and the
#           file dialog (include/sxn/sxnfiledlg.h) with the icons by type it
#           shows and their loader from the profile (mime_icon_load); it needs
#           the POSIX runtime's gui group
core    system/runtime/sxn_runtime.c
core    system/runtime/sxnprofile.c
core    system/runtime/sxn_sxe.c
core    system/runtime/sxndns_message.c
core    system/runtime/sxndns.c
core    system/runtime/sxnhttp_message.c
core    system/runtime/sxnhttp.c
gui     system/runtime/sxngui.c
gui     system/runtime/sxnfiledlg.c
gui     system/gfx/mime_icon.c
gui     system/runtime/sxn_mime_icon.c
